## 🔧 Python 讀取指令（所有終端機通用）

```python
import json

# 讀取集中檔案
with open('distill/terminalB_large_messages.jsonl', 'r') as f:
    messages = [json.loads(line) for line in f]

# 篩選特定檔案的訊息（以 session_id 前綴判斷）
target_prefix = '12836ad8'  # 改成你要處理的檔案前綴
filtered = [m for m in messages if m.get('session_id', '').startswith(target_prefix)]

# 查看有多少則
print(f"找到 {len(filtered)} 則教練訊息")

# 查看所有 session_id（幫助了解結構）
session_ids = list(set(m.get('session_id', '') for m in filtered))
print(f"Session 數量: {len(session_ids)}")
for sid in session_ids:
    count = len([m for m in filtered if m.get('session_id') == sid])
    print(f"  - {sid}: {count}則")
```
