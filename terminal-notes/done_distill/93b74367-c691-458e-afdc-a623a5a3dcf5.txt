sudo launchctl bootout system/com.apple.mediaanalysisd 2>/dev/null || true
sudo launchctl bootout system/com.apple.mds_stores 2>/dev/null || true
sudo launchctl disable system/com.apple.rapportd
sudo launchctl disable system/com.apple.mediaremoteagent
sudo launchctl disable system/com.apple.weatherd
sudo launchctl disable gui/$(id -u)/com.apple.iCloudSearchPartyUserAgent
sudo launchctl disable gui/$(id -u)/com.apple.findmydeviced
sudo launchctl disable gui/$(id -u)/com.apple.findmylocateagent

echo "已停用 8 個主要吃記憶體背景服務"

# 4. 立即釋放記憶體 + 清理快取
sudo purge
sudo rm -rf /Library/Caches/* ~/Library/Caches/* 2>/dev/null
echo "快取已清理，記憶體應已釋放"

# 5. 執行後驗證（執行完立刻跑）

echo "對比完成！差異請看 ~/memory_before.txt 與 ~/memory_after.txt"
