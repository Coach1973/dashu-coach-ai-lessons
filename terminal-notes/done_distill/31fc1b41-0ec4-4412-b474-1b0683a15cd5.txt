    #!/bin/bash                                                                                                         
    check_and_fix_memsearch() {                                                                                         
      local NAME=$1                                                                                                     
      local DIR=$2                                                                                                      
                                                                                                                        
      echo ""                                                                                                           
      echo "===== $NAME ($DIR) ====="                                                                                   
                                                                                                                        
      # 1. extension 結構                                                                                               
      if [ -f "$DIR/extensions/memsearch/index.ts" ]; then                                                              
        echo "[OK] extension index.ts 存在"                                                                             
      elif [ -f "$DIR/extensions/memsearch/memsearch/index.ts" ]; then                                                  
        echo "[FIX] 雙層目錄問題，正在修復..."                                                                          
        cp -r "$DIR/extensions/memsearch/memsearch/." "$DIR/extensions/memsearch/"                                      
        rm -rf "$DIR/extensions/memsearch/memsearch"                                                                    
        echo "[FIXED] 雙層目錄已修復"                                                                                   
      else                                                                                                              
        echo "[MISSING] memsearch extension 不存在，需要手動安裝"                                                       
      fi                                                                                                                
                                                                                                                        
      # 2. openclaw.json memsearch 設定                                                                                 
      if grep -qi memsearch "$DIR/openclaw.json" 2>/dev/null; then                                                      
        echo "[OK] openclaw.json 有 memsearch 設定"                                                                     
      else                                                                                                              
        echo "[MISSING] openclaw.json 缺少 memsearch 設定，請人工確認後補入"                                            
      fi                                                                                                                
                                                                                                                        
      # 3. .memsearch 記憶檔                                                                                            
      MEM_COUNT=$(ls "$DIR/workspace/.memsearch/memory/" 2>/dev/null | wc -l | tr -d ' ')                               
      echo "[INFO] .memsearch/memory/ 有 $MEM_COUNT 個檔案"                                                             
                                                                                                                        
      # 4. 向量索引                                                                                                     
      WORKSPACE="$DIR/workspace"                                                                                        
      if [ -f ~/mac-openclaw-workflows/scripts/derive-collection.sh ]; then                                             
        COLL=$(bash ~/mac-openclaw-workflows/scripts/derive-collection.sh "$WORKSPACE" 2>/dev/null)                     
      else                                                                                                              
        COLL="ms_openclaw_default"                                                                                      
      fi                                                                                                                
      echo "[INFO] collection 名稱：$COLL"                                                                              
                                                                                                                        
      STATS=$(uvx --from 'memsearch[onnx]' memsearch stats --collection "$COLL" 2>&1)                                   
      if echo "$STATS" | grep -q "0 chunks\|not found\|error"; then                                                     
        echo "[FIX] 向量索引空或不存在，正在重建（此步驟需數分鐘）..."                                                  
        uvx --from 'memsearch[onnx]' memsearch index --provider onnx \                                                  
          --collection "$COLL" \                                                                                        
          "$WORKSPACE/.memsearch/memory/" && echo "[FIXED] 索引重建完成" || echo "[ERROR] 索引重建失敗，請回報"         
      else                                                                                                              
        echo "[OK] 向量索引正常"                                                                                        
        echo "$STATS" | head -3                                                                                         
      fi                                                                                                                
    }                                                                                                                   
                                                                                                                        
    check_and_fix_memsearch "1號機（學長/本體）" ~/.openclaw
